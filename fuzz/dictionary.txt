# Stratavm format dictionary for libFuzzer.
# Contains: file magic, section tags, varint patterns, opcode bytes,
# atom/kind/field constants from the default environment, and common
# structured-input tokens.

# File magic and version (8 + 2 bytes)
"STRATAVM"
"\x01\x00"

# Section tags (little-endian u32)
"\x41\x54\x4f\x4d"  # ATOM
"\x53\x43\x4d\x41"  # SCMA
"\x48\x45\x41\x50"  # HEAP
"\x4a\x52\x4e\x4c"  # JRNL
"\x58\x52\x45\x46"  # XREF
"\x4d\x45\x54\x41"  # META

# Journal opcodes (single bytes)
"\x01"   # NewNode
"\x02"   # SetField
"\x03"   # BeginGroup
"\x04"   # EndGroup
"\x05"   # AddChild
"\x06"   # LinkRef
"\x07"   # DropNode
"\x08"   # Snapshot
"\x09"   # SetMeta
"\x0a"   # Nop

# Value type bytes
"\x00"   # Int
"\x01"   # Float
"\x02"   # Atom
"\x03"   # Ref
"\x04"   # List

# Varint encodings for small counts and ids used in seeds
"\x00"   # 0
"\x01"   # 1
"\x02"   # 2
"\x03"   # 3
"\x04"   # 4
"\x05"   # 5
"\x0a"   # 10
"\x7f"   # 127 (max single-byte varint)
"\x80\x01"  # 128 (first 2-byte varint)
"\xff\x7f"  # 16383

# Atom ids for the default env (0-17, single-byte varints)
# 0=document 1=section 2=paragraph 3=link 4=image
# 5=title 6=body 7=order 8=parent 9=href 10=target 11=width 12=height
# 13=root 14=main 15=intro 16=appendix 17=figure

# Kind ids in default env
"\x00"  # document
"\x01"  # section
"\x02"  # paragraph
"\x03"  # link
"\x04"  # image

# Field flags
"\x00"   # kFieldNone
"\x01"   # kFieldRequired
"\x02"   # kFieldIndexed

# IEEE-754 doubles: 0, 1, -1, NaN, Inf
"\x00\x00\x00\x00\x00\x00\x00\x00"
"\x00\x00\x00\x00\x00\x00\xf0\x3f"
"\x00\x00\x00\x00\x00\x00\xf0\xbf"
"\x00\x00\x00\x00\x00\x00\xf8\x7f"
"\x00\x00\x00\x00\x00\x00\xf0\x7f"

# Header flags
"\x00\x00"   # kFlagNone
"\x01\x00"   # kFlagHasXref
"\x02\x00"   # kFlagStrict
"\x03\x00"   # both

# Reserve hint values to stress the group realloc path
"\x00"       # 0 (forces realloc on any forward-ref)
"\x01"       # 1 (triggers bug with >=2 forward-refs)
"\x02"       # 2
"\xff\x7f"   # 16383 (large hint)

# Common atom strings as raw bytes (useful for LP-string mutations)
"document"
"section"
"paragraph"
"link"
"image"
"title"
"body"
"order"
"parent"
"href"
"target"
"width"
"height"
"root"
"main"
"intro"
"appendix"
"figure"
